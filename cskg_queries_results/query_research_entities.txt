SELECT DISTINCT ?entity ?predicate ?type WHERE {
  VALUES ?type {
    cskg-ont:Method
    cskg-ont:Material
    cskg-ont:Metric
    cskg-ont:Task
    cskg-ont:OtherEntity
  }
  ?entity a ?type .
 BIND(rdf:type AS ?predicate)
}
