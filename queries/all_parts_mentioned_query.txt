PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX rpo: <http://www.semanticweb.org/ftsdemo/ontologies/2025/5/rpo#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX owl: <http://www.w3.org/2002/07/owl#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX prov: <http://www.w3.org/ns/prov#>

SELECT ?org ?type (COUNT(DISTINCT ?paper) AS ?numPapers)
WHERE {
    ?paper rpo:mentions ?mentionedEntity .

    {
        ?mentionedEntity rpo:part_of ?org .
    }
    UNION
    {
        ?mentionedEntity rpo:related_to ?org .
    }

    ?org rdf:type ?type .
}
GROUP BY ?org ?type
ORDER BY DESC(?numPapers)
